# Manga MVP Build — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build all remaining MVP screens and features: Transactions (create, list, detail, edit, delete), Goals (list, detail, edit, movements), Categories (list, detail, edit), Settings, and Manga Insights — using the existing DS library and Supabase backend.

**Architecture:** Next.js 15 App Router with server components for data fetching and client components for interactivity. Supabase RPCs for computed data, direct table queries for CRUD. All UI from `@matsborges/manga-design-system`. No global state — each page fetches its own data. Revalidation via `router.refresh()` after mutations.

**Tech Stack:** Next.js 15, TypeScript, Tailwind CSS 4, Supabase (Postgres + Auth + RLS), `@matsborges/manga-design-system@0.1.0`

**Existing patterns to follow:**
- Server page (`page.tsx`): auth check, fetch data, pass props to client component
- Client component (`*-content.tsx`): receives data as props, handles UI + mutations
- Supabase client: `createClient()` from `@/lib/supabase/client` (browser) or `@/lib/supabase/server` (server)
- Currency: `formatCurrency()`, `formatCurrencyInput()`, `parseCurrencyInput()` from `@/lib/utils/masks`
- Commits in Portuguese, code in English

**DS Components available (not all used yet):**
- `TransactionItem` — title, amount, kind (income/expense), description, leading icon
- `GoalCard` — title, value, target, identity (icon/color), status, supportingInformation
- `CategoryItem` — title, icon, mode (display/navigation/selection/spending), spent, limit
- `FinancialCard` — title, value, metrics, hidden toggle
- `FinancialSummary` — income, expense, result, period, mode (simple/default/detailed)
- `BalanceCard` — title, value, variant (default/subtle/brand)
- `MangaInsights` — title, description, used, limit, action, state
- `EmptyState` — title, description, action
- `Dialog` — confirmations
- `Drawer` — bottom sheets
- `Select` — dropdowns
- `Tabs` / `SegmentedControl` — tab switching
- `Toast` — feedback
- `Alert` — inline messages

```tsx
// src/app/(app)/transacoes/nova/new-transaction-form.tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { Button } from "@matsborges/manga-design-system/button";
import { TextField } from "@matsborges/manga-design-system/text-field";
import { TextArea } from "@matsborges/manga-design-system/text-area";
import { Select } from "@matsborges/manga-design-system/select";
import { ArrowBackIcon } from "@matsborges/manga-design-system/icons";
import { formatCurrencyInput, parseCurrencyInput } from "@/lib/utils/masks";
import { createClient } from "@/lib/supabase/client";

type Category = { id: string; name: string; icon: string | null };

export function NewTransactionForm({ categories }: { categories: Category[] }) {
  const router = useRouter();
  const [type, setType] = useState<"income" | "expense">("expense");
  const [amountInput, setAmountInput] = useState("");
  const [description, setDescription] = useState("");
  const [date, setDate] = useState(() => new Date().toISOString().split("T")[0]);
  const [categoryId, setCategoryId] = useState("");
  const [note, setNote] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  const amount = parseCurrencyInput(amountInput);

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    if (amount <= 0) { setError("Informe um valor maior que zero."); return; }
    if (!description.trim()) { setError("Informe uma descrição."); return; }

    setLoading(true);
    setError(null);

    const supabase = createClient();
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) { router.push("/login"); return; }

    const { error: insertError } = await supabase.from("transactions").insert({
      user_id: user.id,
      type,
      amount,
      description: description.trim(),
      date,
      category_id: type === "expense" && categoryId ? categoryId : null,
    });

    if (insertError) {
      setError("Erro ao salvar. Tente novamente.");
      setLoading(false);
      return;
    }

    router.push("/home");
    router.refresh();
  }

  return (
    <div className="flex flex-col bg-manga-background-default" style={{ minHeight: "100dvh" }}>
      {/* Header */}
      <div className="shrink-0" style={{ height: "24px" }} />
      <div className="shrink-0 flex items-center" style={{ height: "64px", padding: "0 16px", gap: "12px" }}>
        <button type="button" onClick={() => router.back()}
          className="flex items-center justify-center text-manga-text-primary"
          style={{ width: "44px", height: "44px" }} aria-label="Voltar">
          <ArrowBackIcon />
        </button>
        <h1 className="font-semibold text-manga-text-primary"
          style={{ fontSize: "20px", lineHeight: "24px" }}>
          Nova Transação
        </h1>
      </div>

      {/* Form */}
      <form onSubmit={handleSubmit} className="flex-1 flex flex-col">
        <div className="flex-1 overflow-y-auto" style={{ padding: "16px" }}>
          <div className="flex flex-col" style={{ gap: "20px" }}>
            {/* Type toggle */}
            <div className="flex" style={{ gap: "8px" }}>
              <button type="button"
                onClick={() => setType("expense")}
                className={`flex-1 font-semibold rounded-lg ${type === "expense" ? "" : "opacity-50"}`}
                style={{
                  height: "44px", fontSize: "14px",
                  backgroundColor: type === "expense" ? "rgb(238,105,31)" : "rgb(241,240,243)",
                  color: type === "expense" ? "white" : "rgb(35,25,56)",
                }}>
                Saída
              </button>
              <button type="button"
                onClick={() => setType("income")}
                className={`flex-1 font-semibold rounded-lg ${type === "income" ? "" : "opacity-50"}`}
                style={{
                  height: "44px", fontSize: "14px",
                  backgroundColor: type === "income" ? "rgb(15,83,37)" : "rgb(241,240,243)",
                  color: type === "income" ? "white" : "rgb(35,25,56)",
                }}>
                Entrada
              </button>
            </div>

            {/* Amount */}
            <TextField label="Valor" type="text" inputMode="numeric" required
              value={amountInput ? `R$ ${amountInput}` : ""}
              onChange={(e) => {
                const raw = e.target.value.replace(/^R\$\s?/, "");
                setAmountInput(formatCurrencyInput(raw));
                setError(null);
              }}
              placeholder="R$ 0,00" />

            {/* Description */}
            <TextField label="Descrição" type="text" required
              value={description}
              onChange={(e) => { setDescription(e.target.value); setError(null); }}
              placeholder={type === "expense" ? "Com o que você gastou?" : "De onde veio?"} />

            {/* Date */}
            <TextField label="Data" type="date" required
              value={date}
              onChange={(e) => setDate(e.target.value)} />

            {/* Category (expense only) */}
            {type === "expense" && categories.length > 0 && (
              <Select label="Categoria (opcional)"
                placeholder="Sem categoria"
                options={categories.map((c) => ({ value: c.id, label: c.name }))}
                value={categoryId}
                onValueChange={setCategoryId} />
            )}

            {/* Note */}
            <TextArea label="Observação (opcional)"
              value={note}
              onChange={(e) => setNote(e.target.value)}
              placeholder="Algum detalhe sobre esta transação?" />

            {error && (
              <p style={{ fontSize: "14px", lineHeight: "21px", color: "#7F1D1D" }}>{error}</p>
            )}
          </div>
        </div>

        {/* Actions */}
        <div className="shrink-0" style={{ padding: "8px 16px" }}>
          <Button type="submit" fullWidth size="large" loading={loading}>
            Registrar {type === "expense" ? "Saída" : "Entrada"}
          </Button>
        </div>
        <div className="shrink-0" style={{ height: "24px" }} />
      </form>
    </div>
  );
}
```

```tsx
// src/app/(app)/transacoes/page.tsx
import { redirect } from "next/navigation";
import { createClient } from "@/lib/supabase/server";
import { TransactionsContent } from "./transactions-content";

export default async function TransactionsPage() {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) redirect("/login");

  const [txResult, catResult, summaryResult] = await Promise.all([
    supabase
      .from("transactions")
      .select("id, type, amount, description, date, category_id, created_at, categories(name, icon)")
      .eq("user_id", user.id)
      .order("date", { ascending: false })
      .order("created_at", { ascending: false })
      .limit(50),
    supabase
      .from("categories")
      .select("id, name")
      .eq("user_id", user.id)
      .eq("is_active", true)
      .order("sort_order"),
    supabase.rpc("get_monthly_summary", { target_user_id: user.id }),
  ]);

  const summary = (summaryResult.data as { income: number; expense: number; balance: number }[] | null)?.[0]
    ?? { income: 0, expense: 0, balance: 0 };

  return (
    <TransactionsContent
      transactions={txResult.data ?? []}
      categories={catResult.data ?? []}
      income={summary.income}
      expense={summary.expense}
    />
  );
}
```

**Step 2: Create client component**

Uses `TransactionItem` from DS. Shows:
- `FinancialSummary` at top (income, expense for current month)

- List of `TransactionItem` grouped by date

```tsx
// src/app/(app)/transacoes/transactions-content.tsx
"use client";

import { useState } from "react";
import { TransactionItem } from "@matsborges/manga-design-system/transaction-item";
import { FinancialSummary } from "@matsborges/manga-design-system/financial-summary";
import { EmptyState } from "@matsborges/manga-design-system/empty-state";
import { ProductIcon } from "@matsborges/manga-design-system/product";
import { Button } from "@matsborges/manga-design-system/button";
import Link from "next/link";
import type { ProductIconName } from "@matsborges/manga-design-system/product";

type Transaction = {
  id: string;
  type: "income" | "expense";
  amount: number;
  description: string;
  date: string;
  category_id: string | null;
  created_at: string;
  categories: { name: string; icon: string | null } | null;
};

type Filter = "all" | "income" | "expense";

export function TransactionsContent({
  transactions,
  categories,
  income,
  expense,
}: {
  transactions: Transaction[];
  categories: { id: string; name: string }[];
  income: number;
  expense: number;
}) {
  const [filter, setFilter] = useState<Filter>("all");

  const filtered = transactions.filter((t) => {
    if (filter === "all") return true;
    return t.type === filter;
  });

  // Group by date
  const grouped = filtered.reduce<Record<string, Transaction[]>>((acc, t) => {
    const key = t.date;
    if (!acc[key]) acc[key] = [];
    acc[key].push(t);
    return acc;
  }, {});

  const dateKeys = Object.keys(grouped).sort((a, b) => b.localeCompare(a));

  function formatDateLabel(dateStr: string) {
    const [y, m, d] = dateStr.split("-").map(Number);
    const date = new Date(y, m - 1, d);
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const yesterday = new Date(today);
    yesterday.setDate(yesterday.getDate() - 1);

    if (date.getTime() === today.getTime()) return "Hoje";
    if (date.getTime() === yesterday.getTime()) return "Ontem";
    return date.toLocaleDateString("pt-BR", { day: "2-digit", month: "long" });
  }

  const filterButtons: { key: Filter; label: string }[] = [
    { key: "all", label: "Todas" },
    { key: "income", label: "Entradas" },
    { key: "expense", label: "Saídas" },
  ];

  return (
    <div className="flex flex-col" style={{ minHeight: "100dvh", paddingBottom: "88px" }}>
      <div style={{ padding: "16px", display: "flex", flexDirection: "column", gap: "16px" }}>
        {/* Summary */}
        <FinancialSummary
          title="Movimentações"
          income={income}
          expense={expense}
          period="Mês atual"
          mode="simple"
        />

        {/* Filters */}
        <div className="flex" style={{ gap: "8px" }}>
          {filterButtons.map((f) => (
            <button key={f.key} type="button"
              onClick={() => setFilter(f.key)}
              className="font-semibold"
              style={{
                padding: "6px 16px",
                borderRadius: "9999px",
                fontSize: "14px",
                backgroundColor: filter === f.key ? "rgb(35,25,56)" : "rgb(241,240,243)",
                color: filter === f.key ? "white" : "rgb(90,82,105)",
              }}>
              {f.label}
            </button>
          ))}
        </div>

        {/* Transaction list */}
        {filtered.length === 0 ? (
          <EmptyState
            title="Sem transações"
            description="Seu histórico começa aqui. Registre sua primeira transação."
            size="compact"
          />
        ) : (
          <div className="flex flex-col" style={{ gap: "16px" }}>
            {dateKeys.map((dateStr) => (
              <div key={dateStr}>
                <p className="font-semibold text-manga-text-secondary"
                  style={{ fontSize: "12px", lineHeight: "16px", marginBottom: "8px", textTransform: "uppercase" }}>
                  {formatDateLabel(dateStr)}
                </p>
                <div className="flex flex-col" style={{ gap: "4px" }}>
                  {grouped[dateStr].map((t) => (
                    <TransactionItem
                      key={t.id}
                      title={t.description}
                      amount={t.amount}
                      kind={t.type}
                      description={t.categories?.name}
                      leading={t.categories?.icon ? (
                        <ProductIcon name={t.categories.icon as ProductIconName} className="w-5 h-5" />
                      ) : undefined}
                      showLeading={!!t.categories?.icon}
                      href={`/transacoes/${t.id}`}
                    />
                  ))}
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
```

## Phase 5 — Manga Insights (Regras basicas)

### Task 11: Insights Engine — regras iniciais

**Step 1: Create the insights engine**

Rule-based, no AI. Evaluates current data and returns up to 3 insights sorted by priority.

```tsx
// src/lib/insights/engine.ts
export type Insight = {
  id: string;
  priority: "P1" | "P2" | "P3"; // attention, opportunity, positive
  title: string;
  description: string;
  used?: number;
  limit?: number;
  action?: { label: string; href: string };
};

export function computeInsights(data: {
  balance: number;
  income: number;
  expense: number;
  categories: { name: string; spent: number; limit: number | null }[];
  goals: { name: string; current_value: number; target: number; status: string }[];
}): Insight[] {
  const insights: Insight[] = [];

  // P1: Category over limit (> 100%)
  for (const cat of data.categories) {
    if (cat.limit && cat.spent > cat.limit) {
      insights.push({
        id: `cat-over-${cat.name}`,
        priority: "P1",
        title: `${cat.name} acima do planejado`,
        description: `Você já gastou R$ ${(cat.spent / 100).toFixed(2).replace(".", ",")} de R$ ${(cat.limit / 100).toFixed(2).replace(".", ",")} planejados.`,
        used: cat.spent,
        limit: cat.limit,
        action: { label: "Ver categoria", href: "/categorias" },
      });
    }
  }

  // P2: Category approaching limit (80-100%)
  for (const cat of data.categories) {
    if (cat.limit && cat.spent >= cat.limit * 0.8 && cat.spent <= cat.limit) {
      insights.push({
        id: `cat-attention-${cat.name}`,
        priority: "P2",
        title: `${cat.name} próximo do planejado`,
        description: `Atenção: você já usou ${Math.round((cat.spent / cat.limit) * 100)}% do planejado.`,
        used: cat.spent,
        limit: cat.limit,
        action: { label: "Ver categoria", href: "/categorias" },
      });
    }
  }

  // P2: Goal needs attention (behind schedule)
  for (const goal of data.goals) {
    if (goal.status === "active" && goal.current_value === 0 && goal.target > 0) {
      insights.push({
        id: `goal-start-${goal.name}`,
        priority: "P2",
        title: `${goal.name} aguardando primeira reserva`,
        description: "Comece a reservar para transformar seu plano em realidade.",
        action: { label: "Ver meta", href: "/metas" },
      });
    }
  }

  // P3: Positive — expenses lower than income
  if (data.income > 0 && data.expense < data.income) {
    insights.push({
      id: "positive-balance",
      priority: "P3",
      title: "Suas contas estão no azul",
      description: `Suas entradas superam as saídas este mês. Continue assim!`,
    });
  }

  // Sort by priority, return max 3
  const priorityOrder = { P1: 0, P2: 1, P3: 2 };
  insights.sort((a, b) => priorityOrder[a.priority] - priorityOrder[b.priority]);
  return insights.slice(0, 3);
}
```

**Step 2: Goal movements RPC for detail page**

```sql
-- docs/migrations/009_goal_detail_rpc.sql
CREATE OR REPLACE FUNCTION public.get_goal_movements(target_goal_id uuid)
RETURNS TABLE(
  id uuid,
  type text,
  amount numeric(12,2),
  created_at timestamptz
) LANGUAGE sql SECURITY DEFINER AS $$
  SELECT gm.id, gm.type, gm.amount, gm.created_at
  FROM public.goal_movements gm
  JOIN public.goals g ON g.id = gm.goal_id
  WHERE gm.goal_id = target_goal_id
    AND g.user_id = auth.uid()
  ORDER BY gm.created_at DESC;
$$;
```

**Step 3: Category spending history RPC (last 3 months)**

```sql
CREATE OR REPLACE FUNCTION public.get_category_spending_history(
  target_category_id uuid,
  months_back int DEFAULT 3
)
RETURNS TABLE(
  month text,
  spent numeric(12,2)
) LANGUAGE sql SECURITY DEFINER AS $$
  SELECT
    to_char(date_trunc('month', t.date), 'YYYY-MM') AS month,
    COALESCE(SUM(t.amount), 0) AS spent
  FROM public.transactions t
  WHERE t.category_id = target_category_id
    AND t.user_id = auth.uid()
    AND t.type = 'expense'
    AND t.date >= (date_trunc('month', CURRENT_DATE) - (months_back || ' months')::interval)
  GROUP BY date_trunc('month', t.date)
  ORDER BY month DESC;
$$;
```
